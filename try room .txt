    <script type="module">
        // Global variables for Firebase environment (MANDATORY, even if not used)
        const appId = typeof __app_id !== 'undefined' ? __app_id : 'default-app-id';
        const firebaseConfig = JSON.parse(typeof __firebase_config !== 'undefined' ? __firebase_config : '{}');
        const initialAuthToken = typeof __initial_auth_token !== 'undefined' ? __initial_auth_token : '';
        console.log(`App ID: ${appId}`);

        // --- DOM Elements ---
        const imageUpload = document.getElementById('image-upload');
        const colorPicker = document.getElementById('color-picker');
        const hexDisplay = document.getElementById('hex-display');
        const generateButton = document.getElementById('generate-button');
        const originalImageDisplay = document.getElementById('original-image-display');
        const resultImageDisplay = document.getElementById('result-image-display');
        const loadingOverlay = document.getElementById('loading-overlay');
        const statusMessage = document.getElementById('status-message');
        const errorBox = document.getElementById('error-box');
        const errorMessage = document.getElementById('error-message');
        const colorPalette = document.getElementById('color-palette');
        const downloadButton = document.getElementById('download-button'); // New

        let originalImageData = { data: null, mimeType: null }; 
        let currentResultImageUrl = null; // New: To store the URL of the generated image

        const MODEL_NAME = 'gemini-2.5-flash-image-preview'; // Model for image editing

        // Updated: Top 20 Versatile Men's Clothing Colors
        const SUGGESTED_COLORS = [
            '#000080', // 1. Navy Blue
            '#36454F', // 2. Charcoal Gray
            '#000000', // 3. Black
            '#FFFFFF', // 4. White
            '#D3D3D3', // 5. Light Gray
            '#C3B091', // 6. Khaki/Tan
            '#800000', // 7. Burgundy/Maroon
            '#228B22', // 8. Forest Green
            '#6B8E23', // 9. Olive Green
            '#87CEEB', // 10. Sky Blue
            '#4169E1', // 11. Royal Blue
            '#E3DAC9', // 12. Stone/Beige
            '#FFFDD0', // 13. Cream/Off-White
            '#5C4033', // 14. Dark Brown
            '#FFDB58', // 15. Mustard Yellow (Accent)
            '#008080', // 16. Teal
            '#C9A9A6', // 17. Dusty Rose
            '#E6E6FA', // 18. Lavender
            '#7DF9FF', // 19. Electric Blue (Bold)
            '#DC143C', // 20. Crimson Red
        ];

        // --- Utility Functions ---

        /**
         * Generic fetch utility with exponential backoff for robustness.
         */
        async function fetchWithRetries(url, options, retries = 3) {
            for (let i = 0; i < retries; i++) {
                try {
                    const response = await fetch(url, options);
                    if (response.ok) {
                        return response;
                    }
                    throw new Error(`HTTP error! Status: ${response.status}`);
                } catch (error) {
                    console.warn(`Attempt ${i + 1} failed: ${error.message}`);
                    if (i === retries - 1) {
                        throw error;
                    }
                    const delay = Math.pow(2, i) * 1000;
                    await new Promise(resolve => setTimeout(resolve, delay));
                }
            }
        }


        // --- Main Logic ---

        /**
         * Calls the Gemini API to perform the image recoloring.
         */
        async function generateColoredImage(base64Image, mimeType, colorHex) {
            hideError();
            loadingOverlay.classList.remove('hidden');
            generateButton.disabled = true;
            downloadButton.classList.add('hidden'); // Hide download button during generation
            statusMessage.textContent = `Applying ${colorHex}... This may take a moment.`;

            const apiKey = ""; // Replace with your actual API Key
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/${MODEL_NAME}:generateContent?key=${apiKey}`;

            // The detailed prompt ensures the AI preserves everything except the clothes' color.
            const prompt = `Keep the person's face, hair, skin tone, posture, facial expression, and background exactly the same. Do not change anything about the body, face, or surroundings. Only change the color of the clothing item (e.g., shirt, dress, jacket) to ${colorHex} while fully preserving the original style, texture, folds, lighting, and shadows. The photo must remain high-resolution, natural, and photorealistic, with the clothing seamlessly updated. Maintain the exact same pose and details as the original.`;

            const payload = {
                contents: [
                    {
                        role: "user",
                        parts: [
                            { text: prompt },
                            {
                                inlineData: {
                                    mimeType: mimeType,
                                    data: base64Image
                                }
                            }
                        ]
                    }
                ],
                generationConfig: {
                    responseModalities: ['TEXT', 'IMAGE']
                },
            };

            try {
                const response = await fetchWithRetries(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });

                const result = await response.json();
                const candidate = result?.candidates?.[0];

                if (candidate) {
                    const imagePart = candidate.content?.parts?.find(p => p.inlineData && p.inlineData.mimeType.startsWith('image/'));

                    if (imagePart) {
                        const base64Data = imagePart.inlineData.data;
                        const imageUrl = `data:image/png;base64,${base64Data}`;
                        
                        currentResultImageUrl = imageUrl; // Store the generated image URL
                        downloadButton.classList.remove('hidden'); // Show download button
                        downloadButton.disabled = false; // Enable download button

                        resultImageDisplay.innerHTML = `<img src="${imageUrl}" class="w-full h-full object-contain rounded-xl" alt="Recolored Dress">`;
                        statusMessage.textContent = `Color ${colorHex} applied successfully!`;
                    } else {
                        throw new Error('AI response did not contain an image. (Check prompt complexity or file size.)');
                    }
                } else {
                    const errorDetails = result.error?.message || 'An unknown API error occurred.';
                    throw new Error(`API failed: ${errorDetails}`);
                }

            } catch (error) {
                console.error("Generation Error:", error);
                showError(`Failed to apply color: ${error.message}.`);
                resultImageDisplay.innerHTML = `<svg class="w-16 h-16 text-gray-400" fill="none" stroke="currentColor" viewBox="0 0 24 24" xmlns="http://www.w3.org/2000/svg"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M7 21h10a2 2 0 002-2V9.414a1 1 0 00-.293-.707l-5.414-5.414A1 1 0 0011.586 3H7a2 2 0 00-2 2v14a2 2 0 002 2z"></path></svg>`;
                downloadButton.classList.add('hidden'); // Hide download button on error
                downloadButton.disabled = true;
                currentResultImageUrl = null;
            } finally {
                loadingOverlay.classList.add('hidden');
                generateButton.disabled = !originalImageData.data;
            }
        }

        // --- UI Handlers ---

        function showError(message) {
            errorBox.classList.remove('hidden');
            errorMessage.textContent = message;
            statusMessage.textContent = 'Generation failed.';
        }

        function hideError() {
            errorBox.classList.add('hidden');
            errorMessage.textContent = '';
        }

        // Initialize the color palette
        function initializeColorPalette() {
            colorPalette.innerHTML = ''; // Clear existing content
            SUGGESTED_COLORS.forEach(color => {
                const swatch = document.createElement('div');
                swatch.className = 'color-swatch';
                swatch.style.backgroundColor = color;
                swatch.setAttribute('data-color', color);
                
                // Set border color for white swatch for visibility
                if (color === '#FFFFFF') {
                    swatch.style.borderColor = '#ccc';
                }

                swatch.addEventListener('click', (e) => {
                    const selectedColor = e.target.getAttribute('data-color');
                    colorPicker.value = selectedColor;
                    hexDisplay.textContent = selectedColor;
                    
                    // Automatically trigger generation upon palette click if an image is loaded
                    if (originalImageData.data) {
                        generateColoredImage(originalImageData.data, originalImageData.mimeType, selectedColor);
                    } else {
                        statusMessage.textContent = 'Image not uploaded. Upload your photo first.';
                    }
                });

                colorPalette.appendChild(swatch);
            });
        }

        // Handle image file selection
        imageUpload.addEventListener('change', async (event) => {
            const file = event.target.files[0];
            if (!file) return;

            hideError();
            downloadButton.classList.add('hidden'); // Hide download button on new upload
            downloadButton.disabled = true;
            currentResultImageUrl = null; // Clear previous result URL

            try {
                const reader = new FileReader();
                reader.onload = (e) => {
                    const imageUrl = e.target.result;
                    originalImageDisplay.innerHTML = `<img src="${imageUrl}" class="w-full h-full object-contain rounded-xl" alt="Original Upload">`;
                    
                    const parts = imageUrl.split(',');
                    const base64Data = parts.pop();
                    const mimeTypeMatch = parts[0].match(/^data:(.+?);base64/);
                    const mimeType = mimeTypeMatch ? mimeTypeMatch[1] : file.type;

                    originalImageData = { data: base64Data, mimeType: mimeType };
                    
                    generateButton.disabled = false;
                    statusMessage.textContent = 'Photo uploaded. Select a color or palette swatch and click "Apply".';
                };
                reader.readAsDataURL(file);

                // Clear result area
                resultImageDisplay.innerHTML = `<svg class="w-16 h-16 text-gray-400" fill="none" stroke="currentColor" viewBox="0 0 24 24" xmlns="http://www.w3.org/2000/svg"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M7 21h10a2 2 0 002-2V9.414a1 1 0 00-.293-.707l-5.414-5.414A1 1 0 0011.586 3H7a2 2 0 00-2 2v14a2 2 0 002 2z"></path></svg>`;

            } catch (error) {
                showError('Could not process the uploaded file.');
                originalImageData = { data: null, mimeType: null };
                generateButton.disabled = true;
            }
        });

        // Handle custom color picker change
        colorPicker.addEventListener('input', (event) => {
            const hex = event.target.value;
            hexDisplay.textContent = hex;
        });

        // Handle the generation button click
        generateButton.addEventListener('click', () => {
            if (!originalImageData.data) {
                showError('Please upload an image first.');
                return;
            }
            const colorHex = colorPicker.value;
            generateColoredImage(originalImageData.data, originalImageData.mimeType, colorHex);
        });

        // Handle download button click (New)
        downloadButton.addEventListener('click', () => {
            if (currentResultImageUrl) {
                const a = document.createElement('a');
                a.href = currentResultImageUrl;
                a.download = 'recolored_outfit.png'; // Suggested filename
                document.body.appendChild(a);
                a.click();
                document.body.removeChild(a);
            } else {
                showError('No recolored image available to download.');
            }
        });

        // Initialize palette on load
        initializeColorPalette();
        
    </script>
